MIDUS Refresher 1 Models: Full-Time, Standard-Shift Sample

About this script

Primary, preregistered analysis (main text). Full-time, standard-daytime-shift participants.

Fits Models 1a, 1a + weekend, 1b, 2a, and 2b (raw TV; preregistered), plus Model 2b with log1p TV (secondary; not preregistered). Run MR_data_structuring.Rmd first. Fitted models, LOO, and k-fold objects are cached in model_fits_mr/fulltime_standard/, so re-knitting reloads them instead of refitting.

Robustness checks (not preregistered): 10-fold cross-validation for the random-scale model comparisons, because many observations have Pareto k > 0.7 even after moment matching; and Model 2b refit with mi() to include days with sleep but missing TV. model_2b_mi uses 4000 iterations (2000 warmup) from the start, because it needed doubled iterations in MIDUS II, and adapt_delta = 0.99 (1 divergent transition in the Refresher at default settings).

Models with doubled iterations (bulk ESS < 1000 at 2000 iterations, per the prereg): model_1b, model_2a, model_2a_tv_complete, model_2b, model_2b_log.

# Computes LOO once and caches it to disk
get_loo <- function(model, path) {
  if (file.exists(path)) {
    readRDS(path)
  } else {
    loo_result <- loo(model, moment_match = TRUE)
    saveRDS(loo_result, path)
    loo_result
  }
}

# kfold() runs refits through the future package, which by default
# refuses to pass objects over 500 MB (the fitted models are larger).
# Folds run one at a time in this session, so nothing is copied.
options(future.globals.maxSize = Inf)

# Computes 10-fold CV once and caches it to disk (refits are not saved,
# to keep disk use low)
get_kfold <- function(model, folds, path) {
  if (file.exists(path)) {
    readRDS(path)
  } else {
    kfold_result <- kfold(model, folds = folds, seed = 2026)
    saveRDS(kfold_result, path)
    kfold_result
  }
}

# Assigns K folds stratified by participant: each person's rows are
# shuffled and dealt across folds in turn, so their days land in different
# folds. Same algorithm as loo::kfold_split_stratified() in loo >= 2.9.0.
# Written out because loo 2.8.0 mishandles participants with a single row
# (sample(n) draws from 1:n), which silently un-stratifies the folds.
split_folds_by_id <- function(id, K = 10) {
  id <- as.integer(factor(id))
  row_order <- integer()
  for (person in seq_len(max(id))) {
    rows <- which(id == person)
    if (length(rows) > 1) rows <- sample(rows)
    row_order <- c(row_order, rows)
  }
  folds <- rep(NA_integer_, length(id))
  folds[row_order] <- rep(1:K, ceiling(length(id) / K))[seq_along(id)]
  folds
}

# Prereg convergence criteria: R-hat < 1.01, bulk and tail ESS > 1000,
# and 0 divergent transitions (population-level, SD, and correlation
# parameters; bsp_ and sigma_ cover the mi() model)
check_convergence <- function(fit) {
  draws_summary <- posterior::summarise_draws(
    as_draws_df(fit, variable = "^(b_|bsp_|sd_|cor_|sigma$|sigma_)", regex = TRUE),
    "rhat", "ess_bulk", "ess_tail"
  )

  tibble(
    max_rhat = max(draws_summary$rhat, na.rm = TRUE),
    min_bulk_ess = min(draws_summary$ess_bulk, na.rm = TRUE),
    min_tail_ess = min(draws_summary$ess_tail, na.rm = TRUE),
    divergent = sum(nuts_params(fit, pars = "divergent__")$Value)
  ) %>%
    mutate(
      meets_prereg = max_rhat < 1.01 & min_bulk_ess > 1000 &
        min_tail_ess > 1000 & divergent == 0
    )
}

# Prereg assumption check: Q-Q plot of level-1 residuals
check_residual_normality <- function(fit, label) {
  resid_est <- residuals(
    fit,
    method = "posterior_predict"
  )[, "Estimate"]

  qqnorm(
    resid_est,
    main = paste0(label, ": Level-1 residual Q-Q plot")
  )

  qqline(resid_est)
}

# Sleep and TV: overall M, SD, and median, plus between-person SD
# (SD of person means) and within-person SD (SD of deviations from
# each person's own mean)
describe_levels <- function(data, variable) {
  data %>%
    filter(!is.na({{ variable }})) %>%
    group_by(id) %>%
    mutate(
      person_mean = mean({{ variable }}),
      deviation = {{ variable }} - person_mean
    ) %>%
    ungroup() %>%
    summarise(
      n = n(),
      mean = mean({{ variable }}),
      sd = sd({{ variable }}),
      median = median({{ variable }}),
      between_person_sd = sd(person_mean[!duplicated(id)]),
      within_person_sd = sd(deviation)
    )
}
